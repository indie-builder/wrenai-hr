  \ >= DATE '2025-01-01'\n    AND termination_date < DATE '2026-01-01'\n), opening\
  \ AS (\n  SELECT COUNT(*) AS n\n  FROM employees\n  WHERE hire_date < DATE '2025-01-01'\n\
  \    AND (termination_date IS NULL OR termination_date >= DATE '2025-01-01')\n)\n\
  SELECT l.n AS 离职人数, h.n AS 期初在职,\n       ROUND(l.n * 100.0 / NULLIF(h.n, 0), 2)\
  \ AS 离职率百分比\nFROM leavers l CROSS JOIN opening h"
