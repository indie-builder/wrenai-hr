nl: 2026H1各部门绩效目标加权平均完成率？
sql: 'SELECT d.dept_name AS 部门, round(sum(g.completion_pct * g.weight) / sum(g.weight),
  1) AS 加权完成率,

  count(DISTINCT g.emp_id) AS 员工数

  FROM performance_goals g JOIN employees e ON g.emp_id = e.emp_id JOIN departments
  d ON d.dept_id = e.dept_id

  WHERE g.review_period = ''2026H1'' GROUP BY d.dept_name'
