nl: 2025年度调薪渗透率（调薪去重人数/期初在职）？
sql: 'WITH sc AS (SELECT count(DISTINCT emp_id) AS n FROM salary_changes WHERE effective_date
  BETWEEN DATE ''2025-01-01'' AND DATE ''2025-12-31''),

  hc AS (SELECT count(*) AS n FROM employees WHERE hire_date < DATE ''2025-01-01''
  AND (termination_date IS NULL OR termination_date >= DATE ''2025-01-01''))

  SELECT sc.n AS 调薪人数, hc.n AS 期初在职, round(sc.n * 100.0 / hc.n, 1) AS 调薪渗透率 FROM sc,
  hc'
