nl: 2025年单人招聘成本（渠道费用合计/新入职人数）？
sql: 'WITH c AS (SELECT coalesce(sum(amount), 0) AS v FROM recruitment_costs WHERE
  cost_month LIKE ''2025%''),

  h AS (SELECT count(*) AS n FROM employees WHERE hire_date BETWEEN DATE ''2025-01-01''
  AND DATE ''2025-12-31'')

  SELECT c.v AS 招聘费用合计, h.n AS 新入职人数, round(c.v / h.n, 0) AS 单人招聘成本 FROM c, h'
