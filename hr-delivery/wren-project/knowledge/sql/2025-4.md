nl: 2025年招聘漏斗：投递→入职转化率？
sql: 'WITH funnel AS (SELECT count(*) AS applied, count(*) FILTER (WHERE stage = ''已入职'')
  AS hired

  FROM candidates WHERE applied_at BETWEEN ''2025-01-01'' AND ''2025-12-31'')

  SELECT applied AS 投递人数, hired AS 入职人数, round(hired * 100.0 / applied, 2) AS 转化率百分比
  FROM funnel'
