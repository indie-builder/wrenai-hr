nl: 2026H1绩效C/D在职员工的部门分布Top5？
sql: 'SELECT d.dept_name AS 部门, count(*) AS 低绩效人数 FROM performance_reviews pr

  JOIN employees e ON pr.emp_id = e.emp_id JOIN departments d ON e.dept_id = d.dept_id

  WHERE pr.review_period = ''2026H1'' AND pr.grade IN (''C'',''D'') AND e.status =
  ''在职'' GROUP BY d.dept_name LIMIT 5'
