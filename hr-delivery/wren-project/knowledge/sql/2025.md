nl: 2025年公司整体离职率（官方口径）？
sql: 'WITH leavers AS (SELECT count(*) AS n FROM employees WHERE termination_date
  >= DATE ''2025-01-01'' AND termination_date < DATE ''2026-01-01''),

  hc AS (SELECT count(*) AS n FROM employees WHERE hire_date < DATE ''2025-01-01''
  AND (termination_date IS NULL OR termination_date >= DATE ''2025-01-01''))

  SELECT l.n AS 离职人数, h.n AS 期初在职, round(l.n * 100.0 / h.n, 2) AS 离职率百分比 FROM leavers
  l, hc h'
