nl: 2024年以来晋升人次与晋升率（期初2024-01-01）？
sql: 'WITH p AS (SELECT count(*) AS n FROM promotions WHERE promo_date >= DATE ''2024-01-01''),

  hc AS (SELECT count(*) AS n FROM employees WHERE hire_date < DATE ''2024-01-01''
  AND (termination_date IS NULL OR termination_date >= DATE ''2024-01-01''))

  SELECT p.n AS 晋升人次, hc.n AS 期初在职, round(p.n * 100.0 / hc.n, 2) AS 晋升率 FROM p, hc'
